\section{Algorithms}

\begin{frame}{Partially Greedy Cyclic Coordinate Descent (PGCCD)}
\texttt{L0Learn} \cite{fastselect} proposes a CD method:
\[
x^{k+1}_i \in \argmin_{u_i} F(x^k + (u_i-x_i^k)e_i)
= \prox_{\frac{1}{\|A_i\|^2+2\lambda_2} (\lambda_0\mathbf{1}_{\{0\}}+\lambda_1|\cdot|)}(\tilde{x_i^k}).
\]
\begin{itemize}
\item Start with $x^0$, residual $r^0 = b - Ax^0$;
\item Reorder coordinates in descending order of $|\langle r^0, A_i \rangle|$;
\item Only the top $t$ ($t \ll p$) are ordered; the rest keep original order;
\item Cyclic updates after the initial ordering.
\end{itemize}
Converges to a CW minimum under weak assumptions.
\end{frame}

\begin{frame}{Nonmonotone Proximal Gradient (NPG)}
Algorithm for composite problems like \eqref{C} \cite{kanzow2021}.
\begin{algorithmic}[1]
\State Input $x^0$, $\gamma_{\max},\gamma_{\min}>0$, $m\in\N$, $\delta,\tau\in(0,1)$.
\Repeat
\State Choose $\gamma_{k,0} \in [\gamma_{\min},\gamma_{\max}]$, $i\gets0$.
\While{$F(x^{k,i}) > \max_{0\le j \le \min\{k,m-1\}} F(x^{k-j}) - \frac{\delta}{2\gamma_{k,i}}\|x^{k,i}-x^k\|^2$}
\State $x^{k,i} \in \prox_{\gamma_{k,i} h}(x^k - \gamma_{k,i}\nabla f(x^k))$
\State $\gamma_{k,i+1} \gets \tau \gamma_{k,i}$, $i \gets i+1$
\EndWhile
\State $x^{k+1} \gets x^{k,i}$, $k \gets k+1$
\Until{stopping criterion}
\end{algorithmic}
Optionally return $x^{\text{best}}$ with smallest $F$.
\end{frame}

\begin{frame}{NSPG – Spectral Step Size}
Using the primal spectral step \cite{BB1,BB2}:
\begin{equation} \label{BBR}
\gamma_k^{BBR} \coloneqq \frac{(s^k)^T s^k}{(s^k)^T r^k},\quad
s^k = x^k - x^{k-1},\; r^k = \nabla f(x^k) - \nabla f(x^{k-1}).
\end{equation}
As in \cite{positiveBB}, the initial step is
\[
\gamma_{k,0} \gets \gamma_{k,+}^{BBR} \coloneqq 
\begin{cases}
\gamma_k^{BBR}, & \gamma_k^{BBR} > 0,\\
\frac{\|s^k\|}{\|r^k\|}, & \text{otherwise}.
\end{cases}
\]
Interpretation: $\prox$ with this step minimizes a second‑order model of $f$ using a scalar Hessian approximation $\mu_k I$ from the secant equation.
\end{frame}

\begin{frame}{NSPGH – Hybrid Spectral Step}
The dual spectral step:
\begin{equation} \label{BBR2}
\gamma_k^{BBR2} \coloneqq \frac{(s^k)^T r^k}{(r^k)^T r^k},\qquad
\gamma_{k,+}^{BBR2} \coloneqq \begin{cases}
\gamma_k^{BBR2}, & \gamma_k^{BBR2} > 0,\\
\frac{\|r^k\|}{\|s^k\|}, & \text{otherwise}.
\end{cases}
\end{equation}
Hybrid choice \cite{hibrido1,hibrido2}:
\[
\gamma^{BBRH}_{k,+} = 
\begin{cases}
\gamma^{BBR2}_{k,+}, & \gamma^{BBR}_{k,+} < \delta \gamma^{BBR2}_{k,+},\\[4pt]
\gamma^{BBR}_{k,+} - \frac{1}{\delta} \gamma^{BBR2}_{k,+}, & \text{otherwise},
\end{cases}
\]
with $\delta=2$ typical.
\end{frame}

\begin{frame}{VMNSPG – Variable Metric Proximal Gradient}
Proximal operator with a metric $U \succ 0$ \cite{vmspg}:
\[
\prox_{g,U}(x) \coloneqq \argmin_{u} g(u) + \frac{1}{2}\|u-x\|_U^2.
\]
Update step:
\[
x^{k,i} \in \prox_{h, U_{k,i}}\bigl(x^k - U_{k,i}^{-1} \nabla f(x^k)\bigr),
\]
with a nonmonotone line search. The diagonal metric $U_{k,+}^{DBBR}$ solves
\[
\min_{U=\operatorname{Diag}(u)} \|U s^k - r^k\|_2^2 + \mu \|U-U_{k-1}\|_F^2,
\]
subject to $\frac{1}{\gamma^{BBR}_{k,+}} I \preceq U \preceq \frac{1}{\gamma^{BBR2}_{k,+}} I$.
Closed-form:
\[
(U_{k,+}^{DBBR})_{ii} = \max\Bigl\{\frac{1}{\gamma^{BBR}_{k,+}},\,
\min\Bigl\{\frac{1}{\gamma^{BBR2}_{k,+}},\,
\frac{(s_i^k)^2 + \mu (U_{k-1})_{ii}^2}{(s_i^k)^2 + \mu}\Bigr\}\Bigr\}.
\]
\end{frame}

\begin{frame}{Local Combinatorial Optimization – CPSI($k$)}
To find a PSI($k$) minimum, alternate between an \textit{inner} solver (e.g., PGCCD, NSPG) and a combinatorial perturbation \cite{fastselect}:
\begin{equation} \label{psik}
\min_{S_1 \subset S, S_2 \subset S^c, |S_1|,|S_2| \le k} \min_{u \in \R^{|S_2|}} F(x^* - U_{S_1}x^*_{S_1} + U_{S_2}u).
\end{equation}
\end{frame}

\begin{frame}{CPSI($k$) Algorithm}
\begin{algorithm}[H]
\caption{CPSI($k$) Method}
\begin{algorithmic}[1]
\Input{$\hat{x}^0$, $k$, inner algorithm}
\State $\ell \gets 0$
\While{true}
\State $\hat{x}^{\ell+1} \gets$ output of inner algorithm starting from $\hat{x}^\ell$
\If{problem \eqref{psik} has feasible $\hat{x}$ with $F(\hat{x}) < F(\hat{x}^{\ell+1})$}
\State $\hat{x}^{\ell+1} \gets \hat{x}$, $\ell \gets \ell+1$
\Else \textbf{break}
\EndIf
\EndWhile
\Output{last $\hat{x}^{\ell+1}$ (PSI($k$) minimum)}
\end{algorithmic}
\end{algorithm}
\end{frame}

\begin{frame}{Efficient PSI(1) Perturbation}
\begin{small}
For $k=1$, the swap search becomes:
\begin{algorithmic}[1]
\For{$i\in S$}
\For{$j\in S^c$}
\State $v_j^* \in \prox_{\frac{1}{\|A_j\|^2+2\lambda_2}(\lambda_0\mathbf{1}_{\{0\}}+\lambda_1|\cdot|)}(\bar{x}_{ij}^\ell)$, where 
$\bar{x}_{ij}^\ell = \frac{\langle r^\ell + x_i^\ell A_i, A_j\rangle}{\|A_j\|^2+2\lambda_2}$
\EndFor
\State $\vartheta \gets \argmax_{j\in S^c} |v_j^*|$
\If{$|v_\vartheta^*| > |x_i^\ell|$}
\State $\hat{x} \gets x^\ell - e_i x_i^\ell + e_\vartheta v_\vartheta^*$, \textbf{break}
\EndIf
\EndFor
\end{algorithmic}
Properties: $\argmin_j F_j^* \iff \argmax_j |v_j^*|$, and $F_\vartheta^* < F(x^\ell) \iff |v_\vartheta^*| > |x_i^\ell|$.
\end{small}
\end{frame}

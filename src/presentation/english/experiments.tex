\section{Numerical Experiments}

\subsection{Experimental Setup}
\begin{frame}{Data Generation}
\begin{itemize}
\item $A_{n\times p} \sim \operatorname{MVN}(0,\Sigma)$, columns normalized to unit $\ell_2$ norm;
\item The ground truth $x^\dagger$ with $k^\dagger$ nonzero entries (values and indices defined per experiment);
\item Target SNR: $\sigma = \sqrt{\|Ax^\dagger\|^2 / (n\cdot\text{SNR})}$;
\item $b = Ax^\dagger + \epsilon$, $\epsilon_i \stackrel{\text{iid}}{\sim} N(0,\sigma^2)$.
\end{itemize}
Covariance structures:
\begin{itemize}
\item \emph{Constant correlation}: $\sigma_{ij} = \rho\ (i\neq j)$, $\sigma_{ii}=1$;
\item \emph{Exponential correlation}: $\sigma_{ij} = \rho^{|i-j|}$.
\end{itemize}
\end{frame}

\begin{frame}{Algorithm Parameters}
\begin{itemize}
\item NPG family: $m=15$, $\delta=0.01$, $\tau=0.25$, $\gamma_{\min},\gamma_{\max}$ small/large;
\item Spectral steps initialized via \eqref{BBR}, \eqref{BBR2} with $x^{-1}=x^0 - 10^{-5}\nabla f(x^0)$;
\item VMNSPG: $\mu=10^{-3}$;
\item PGCCD: only the top quarter of coordinates are reordered.
\end{itemize}
\end{frame}

\subsection{Comparison: PGCCD vs. NPG Variants}
\begin{frame}{Experiment $\ell_0$ Problem ($\lambda_q=0$)}
Exponential correlation, $\rho=0.5$, $n=500$, $p=2000$, $k^\dagger=100$, SNR$=10$,\\
$\lambda_0 = 0.5\,\|A^T b\|_\infty^2/(2 L_f)$.
\begin{itemize}
\item 50 random problem instances: $x^\dagger$ support and values in $(0,1)$ drawn uniformly;
\item Algorithms start from random initializations with the same support;
\item Support recovery measure:
\[
\frac{|\Supp(x) \cap \Supp(x^\dagger)|}{\max\{|\Supp(x)|, k^\dagger\}};
\]
\item Stopping criterion: relative objective change $< 10^{-7}$.
\end{itemize}
\end{frame}

\subsection{Cross-Validation Strategies}
\begin{frame}{Overview}
Three cross-validation (CV) strategies were implemented to select $\lambda_0$:
\begin{enumerate}
\item \textbf{Standard Decreasing}: Warm-start from a large $\lambda$, reduce monotonically;
\item \textbf{Inverse Increasing}: Warm-start from a small $\lambda$, increase monotonically;
\item \textbf{Smart Adaptive}: Probes both extremes, chooses best direction, allows one reversal.
\end{enumerate}
After CV, a \emph{Final Refinement} compares path-based and zero-based (cold-start) solutions and selects the one with lower validation MSE.
\end{frame}

\begin{frame}{Update Rules for $\lambda$}
\begin{itemize}
    \item \textbf{Decreasing Direction} (Next entry candidate):
    \[
    \lambda_{j+1} = 0.9 \cdot \min\left(\lambda_j, \gamma_{j,0} \cdot \frac{\max_{l \in S^c} |\nabla f(x^{j,0})_l|^2}{2}\right)
    \]
    Finds the largest $\lambda$ that would likely allow a new variable into the support $S$ via the proximal step.
    \item \textbf{Increasing Direction} (Next removal candidate):
    \[
    \lambda_{j+1} = 1.11 \cdot \max\left(\lambda_j, \frac{\min_{l \in S} |x^{j,0}_l - \gamma_{j,0} \nabla f(x^{j,0})_l|^2}{2\gamma_{j,0}}\right)
    \]
    Finds the smallest $\lambda$ that would likely force an active variable out of the support.
\end{itemize}
\end{frame}

\subsection{Tridiagonal Block Experiments}
\begin{frame}{Motivation}
We mask the design matrix $A$ to have a nearly tridiagonal block structure ($\approx 3$ nonzeros per row after balancing). 
This isolates blocks of variables, retaining local covariance $\Sigma$ but removing global dense connections.
\end{frame}
